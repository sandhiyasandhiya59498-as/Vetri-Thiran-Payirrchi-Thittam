from fastapi import FastAPI, Request from fastapi.templating import Jinja2Templates from fastapi.responses import HTMLResponse from app.routes import router app = FastAPI(title="ComicCraft AI") templates = Jinja2Templates(directory="app/templates") app.include_router(router) @app.get("/", response_class=HTMLResponse) def home(request: Request): return templates.TemplateResponse("index.html", {"request": request})
